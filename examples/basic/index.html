    Basic Usage Examples

    This section provides practical examples for getting started with QuantumMetaGPT's core functionality.

    Example 1: Simple Quantum Circuit Generation

    Generate a basic quantum circuit using a trained RL agent:

    import qmetagpt
import numpy as np

# Initialize PPO agent
agent = qmetagpt.PPOAgent(
    state_dim=8,
    action_dim=12,
    policy='MlpPolicy'
)

# Build the model
agent.build_model(
    policy_kwargs=dict(
        net_arch=[64, 64],
        activation_fn='tanh'
    )
)

# Define problem state (example: creating Bell state)
task_state = np.array([1, 0, 1, 0, 0, 1, 1, 0])

# Generate quantum circuit
circuit = agent.generate_circuit(task_state)

print("Generated Quantum Circuit:")
print(circuit)

     └───┘┌─┴─┐

    Example 2: Circuit Evaluation and Analysis

    Evaluate the performance of generated quantum circuits:

    import qmetagpt

# Create evaluator for simulation
evaluator = qmetagpt.QuantumEvaluator(
    use_hardware=False,
    backend_name='aer_simulator'
)

# Evaluate the circuit
results = evaluator.evaluate(
    circuit=circuit,
    shots=1024
)

# Print evaluation results
print(f"Fidelity: {results.get('fidelity', 'N/A'):.4f}")
print(f"Success Probability: {results.get('success_prob', 'N/A'):.4f}")
print(f"Execution Time: {results.get('execution_time', 'N/A'):.3f} seconds")
print(f"Circuit Depth: {results.get('depth', 'N/A')}")

# Display measurement counts
if 'counts' in results:
    print("\nMeasurement Results:")
    for state, count in results['counts'].items():
        probability = count / sum(results['counts'].values())
        print(f"  |{state}⟩: {count} ({probability:.3f})")

    Example 3: Batch Processing Multiple Tasks

    Process multiple quantum problems efficiently:

    import qmetagpt
import numpy as np

def batch_quantum_processing():
    # Initialize agent
    agent = qmetagpt.PPOAgent(state_dim=8, action_dim=12)
    agent.build_model()

    # Initialize evaluator
    evaluator = qmetagpt.QuantumEvaluator()

    # Define multiple tasks
    tasks = [
        np.array([1, 0, 1, 0, 0, 1, 1, 0]),  # Bell state
        np.array([0, 1, 0, 1, 1, 0, 0, 1]),  # GHZ state
        np.array([1, 1, 0, 0, 1, 1, 0, 0]),  # W state
    ]

    task_names = ["Bell State", "GHZ State", "W State"]
    results = []

    print("Processing quantum tasks...")
    print("=" * 50)

    for i, (task, name) in enumerate(zip(tasks, task_names)):
        print(f"Task {i+1}: {name}")

        # Generate circuit
        circuit = agent.generate_circuit(task)

        # Evaluate circuit
        result = evaluator.evaluate(circuit, shots=1024)
        result['task_name'] = name
        result['task_id'] = i + 1
        results.append(result)

        print(f"  Fidelity: {result.get('fidelity', 'N/A'):.4f}")
        print(f"  Depth: {result.get('depth', 'N/A')}")
        print()

    return results

# Run batch processing
batch_results = batch_quantum_processing()

# Generate summary
print("Batch Processing Summary:")
print("=" * 50)
best_fidelity = max(r.get('fidelity', 0) for r in batch_results)
avg_fidelity = sum(r.get('fidelity', 0) for r in batch_results) / len(batch_results)

print(f"Best Fidelity: {best_fidelity:.4f}")
print(f"Average Fidelity: {avg_fidelity:.4f}")

    Example 4: Custom Training Environment

    Create and train an agent on custom quantum tasks:

    import qmetagpt
import numpy as np

# Define custom quantum environment
class CustomQuantumEnv:
    def __init__(self, target_state="bell"):
        self.target_state = target_state
        self.n_qubits = 2
        self.max_steps = 10
        self.current_step = 0

    def reset(self):
        self.current_step = 0
        return np.zeros(8)  # Initial state

    def step(self, action):
        self.current_step += 1

        # Convert action to quantum gate
        reward = self._calculate_reward(action)
        done = self.current_step >= self.max_steps
        next_state = self._get_next_state(action)

        return next_state, reward, done, {}

    def _calculate_reward(self, action):
        # Reward based on proximity to target state
        if self.target_state == "bell":
            return np.random.uniform(0.8, 1.0)  # Simplified reward
        return np.random.uniform(0.5, 0.9)

    def _get_next_state(self, action):
        return np.random.random(8)  # Simplified state transition

# Train agent on custom environment
def train_custom_agent():
    # Create environment
    env = CustomQuantumEnv(target_state="bell")

    # Initialize agent
    agent = qmetagpt.A2CAgent(
        state_dim=8,
        action_dim=6,
        learning_rate=0.0003
    )

    agent.build_model(
        policy_kwargs=dict(
            net_arch=[64, 32],
            activation_fn='relu'
        )
    )

    print("Training agent on custom environment...")

    # Train the agent
    agent.train(env, total_timesteps=10000)

    print("Training completed!")

    # Test trained agent
    test_state = env.reset()
    circuit = agent.generate_circuit(test_state)

    print(f"Generated circuit after training:")
    print(circuit)

    return agent

# Run custom training
trained_agent = train_custom_agent()

    Example 5: Report Generation

    Generate comprehensive reports for your quantum algorithms:

    import qmetagpt

def generate_analysis_report():
    # Initialize components
    agent = qmetagpt.PPOAgent(state_dim=8, action_dim=12)
    agent.build_model()

    evaluator = qmetagpt.QuantumEvaluator()
    report_builder = qmetagpt.ReportBuilder()

    # Generate circuit
    task_state = np.array([1, 0, 1, 0, 0, 1, 1, 0])
    circuit = agent.generate_circuit(task_state)

    # Evaluate performance
    results = evaluator.evaluate(circuit, shots=2048)

    # Add metadata
    metadata = {
        'algorithm_type': 'PPO-generated',
        'target_task': 'Bell state preparation',
        'training_timesteps': 10000,
        'evaluation_shots': 2048,
        'backend': 'aer_simulator'
    }

    # Generate comprehensive report
    report_builder.generate_analysis_report(
        results=results,
        circuit=circuit,
        metadata=metadata,
        output_file="quantum_analysis_report.pdf",
        include_visualizations=True
    )

    print("✅ Analysis report generated: quantum_analysis_report.pdf")

    # Generate summary statistics
    print("\nSummary Statistics:")
    print(f"Circuit Fidelity: {results.get('fidelity', 'N/A'):.4f}")
    print(f"Gate Count: {results.get('gate_count', 'N/A')}")
    print(f"Circuit Depth: {results.get('depth', 'N/A')}")
    print(f"Execution Time: {results.get('execution_time', 'N/A'):.3f}s")

# Generate report
generate_analysis_report()

    Example 6: Error Handling and Validation

    Robust error handling for production use:

    import qmetagpt
import logging

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def robust_quantum_processing(task_states, max_retries=3):
    """
    Process quantum tasks with robust error handling
    """
    try:
        # Verify license first
        machine_id = qmetagpt.get_machine_id()
        logger.info(f"Processing with Machine ID: {machine_id}")

        # Initialize components with error handling
        agent = None
        evaluator = None

        try:
            agent = qmetagpt.PPOAgent(state_dim=8, action_dim=12)
            agent.build_model()
            logger.info("✅ Agent initialized successfully")
        except Exception as e:
            logger.error(f"❌ Agent initialization failed: {e}")
            raise

        try:
            evaluator = qmetagpt.QuantumEvaluator()
            logger.info("✅ Evaluator initialized successfully")
        except Exception as e:
            logger.error(f"❌ Evaluator initialization failed: {e}")
            raise

        results = []

        for i, task_state in enumerate(task_states):
            retries = 0
            while retries < max_retries:
                try:
                    logger.info(f"Processing task {i+1}/{len(task_states)} (attempt {retries+1})")

                    # Validate input
                    if not isinstance(task_state, np.ndarray) or task_state.shape != (8,):
                        raise ValueError(f"Invalid task state shape: {task_state.shape}")

                    # Generate circuit
                    circuit = agent.generate_circuit(task_state)

                    if circuit is None:
                        raise RuntimeError("Circuit generation returned None")

                    # Evaluate circuit
                    result = evaluator.evaluate(circuit, shots=1024)

                    if result is None:
                        raise RuntimeError("Circuit evaluation returned None")

                    result['task_id'] = i
                    result['attempts'] = retries + 1
                    results.append(result)

                    logger.info(f"✅ Task {i+1} completed successfully")
                    break

                except Exception as e:
                    retries += 1
                    logger.warning(f"⚠️ Task {i+1} attempt {retries} failed: {e}")

                    if retries >= max_retries:
                        logger.error(f"❌ Task {i+1} failed after {max_retries} attempts")
                        # Add failed task to results with error info
                        results.append({
                            'task_id': i,
                            'error': str(e),
                            'attempts': retries,
                            'status': 'failed'
                        })

        return results

    except qmetagpt.LicenseEnforcementError as e:
        logger.error(f"❌ License error: {e}")
        logger.error(f"Machine ID: {qmetagpt.get_machine_id()}")
        logger.error("Contact: bajpaikrishna715@gmail.com")
        raise
    except Exception as e:
        logger.error(f"❌ Unexpected error: {e}")
        raise

# Example usage with error handling
if __name__ == "__main__":
    # Define test tasks
    test_tasks = [
        np.array([1, 0, 1, 0, 0, 1, 1, 0]),  # Valid task
        np.array([0, 1, 0, 1, 1, 0, 0, 1]),  # Valid task
        np.array([1, 1, 0, 0]),              # Invalid shape (will cause error)
        np.array([1, 1, 0, 0, 1, 1, 0, 0]),  # Valid task
    ]

    try:
        results = robust_quantum_processing(test_tasks)

        # Print results summary
        successful_tasks = [r for r in results if r.get('status') != 'failed']
        failed_tasks = [r for r in results if r.get('status') == 'failed']

        print(f"\nProcessing Complete:")
        print(f"✅ Successful: {len(successful_tasks)}")
        print(f"❌ Failed: {len(failed_tasks)}")

        if successful_tasks:
            avg_fidelity = sum(r.get('fidelity', 0) for r in successful_tasks) / len(successful_tasks)
            print(f"📊 Average Fidelity: {avg_fidelity:.4f}")

    except Exception as e:
        print(f"❌ Critical error: {e}")

    Common Patterns

    Pattern 1: Initialize-Process-Report

    # Standard workflow pattern
agent = qmetagpt.PPOAgent(state_dim=8, action_dim=12)
agent.build_model()

evaluator = qmetagpt.QuantumEvaluator()
circuit = agent.generate_circuit(task)
results = evaluator.evaluate(circuit)

report = qmetagpt.ReportBuilder()
report.generate_analysis_report(results, "output.pdf")

    Pattern 2: Batch Processing with Progress

    # Batch processing with progress tracking
from tqdm import tqdm

tasks = [...]  # Your tasks
results = []

for task in tqdm(tasks, desc="Processing quantum tasks"):
    circuit = agent.generate_circuit(task)
    result = evaluator.evaluate(circuit)
    results.append(result)

    Pattern 3: Configuration-Based Setup

    # Configuration-driven initialization
config = {
    'agent': {
        'type': 'PPO',
        'state_dim': 8,
        'action_dim': 12,
        'policy_kwargs': {'net_arch': [64, 64]}
    },
    'evaluator': {
        'use_hardware': False,
        'backend_name': 'aer_simulator',
        'shots': 1024
    }
}

agent = qmetagpt.get_agent(**config['agent'])
evaluator = qmetagpt.QuantumEvaluator(**config['evaluator'])
