    Quick Start Guide

    Get up and running with QuantumMetaGPT in under 10 minutes!

    Prerequisites

    • QuantumMetaGPT installed (Installation Guide)
    • Valid license activated
    • Python 3.10+ environment

    Your First Quantum Algorithm

    Step 1: Import and Check License

    import qmetagpt
import numpy as np

# Verify license and get machine info
print("🔧 Machine ID:", qmetagpt.get_machine_id())
qmetagpt.show_license_info()

    Step 2: Initialize RL Agent

    # Create a PPO agent for quantum circuit generation
agent = qmetagpt.PPOAgent(
    state_dim=8,        # Problem state dimensions
    action_dim=12,      # Quantum gate action space
    policy='MlpPolicy'  # Neural network policy
)

# Build the reinforcement learning model
agent.build_model(
    policy_kwargs=dict(
        net_arch=[64, 64],  # Neural network architecture
        activation_fn='relu'
    )
)

print("✅ PPO Agent initialized successfully!")

    Step 3: Generate Quantum Circuit

    # Define a simple quantum task (e.g., creating entanglement)
task_state = np.array([1, 0, 1, 0, 0, 1, 1, 0])  # Problem encoding

# Generate quantum circuit using trained agent
quantum_circuit = agent.generate_circuit(task_state)

print("🔬 Generated Quantum Circuit:")
print(quantum_circuit)

    Step 4: Evaluate Performance

    # Initialize quantum evaluator
evaluator = qmetagpt.QuantumEvaluator(
    use_hardware=False,  # Use simulator for quick start
    backend_name='aer_simulator'
)

# Evaluate circuit performance
results = evaluator.evaluate(
    circuit=quantum_circuit,
    shots=1024
)

print("📊 Evaluation Results:")
print(f"Fidelity: {results.get('fidelity', 'N/A')}")
print(f"Success Probability: {results.get('success_prob', 'N/A')}")
print(f"Execution Time: {results.get('execution_time', 'N/A')} seconds")

    Step 5: Generate Report

    # Create comprehensive report
report_builder = qmetagpt.ReportBuilder()

# Generate analysis report
report_builder.generate_analysis_report(
    results=results,
    circuit=quantum_circuit,
    output_file="my_first_quantum_algorithm.pdf"
)

print("📄 Report generated: my_first_quantum_algorithm.pdf")

    Complete Example

    Here's the complete example in one script:

    #!/usr/bin/env python3
"""
QuantumMetaGPT Quick Start Example
Generates and evaluates a quantum algorithm using RL
"""

import qmetagpt
import numpy as np

def main():
    print("🚀 QuantumMetaGPT Quick Start")
    print("=" * 50)

    # Step 1: Verify license
    print("🔐 Checking license...")
    qmetagpt.show_license_info()

    # Step 2: Initialize agent
    print("🤖 Initializing PPO Agent...")
    agent = qmetagpt.PPOAgent(state_dim=8, action_dim=12)
    agent.build_model(policy_kwargs=dict(net_arch=[64, 64]))

    # Step 3: Generate circuit
    print("⚡ Generating quantum circuit...")
    task_state = np.random.random(8)  # Random task encoding
    circuit = agent.generate_circuit(task_state)

    # Step 4: Evaluate
    print("🔬 Evaluating circuit...")
    evaluator = qmetagpt.QuantumEvaluator()
    results = evaluator.evaluate(circuit, shots=1024)

    # Step 5: Report
    print("📊 Generating report...")
    report = qmetagpt.ReportBuilder()
    report.generate_analysis_report(results, circuit, "quickstart_results.pdf")

    print("✅ Quick start completed successfully!")
    print("📄 Check 'quickstart_results.pdf' for detailed analysis")

if __name__ == "__main__":
    main()

    Advanced Examples

    Custom RL Training

    # Train agent on specific quantum tasks
import qmetagpt

# Create training environment
env = qmetagpt.QuantumCircuitEnv(
    n_qubits=4,
    max_gates=20,
    target_state="bell_state"
)

# Initialize and train agent
agent = qmetagpt.A2CAgent(state_dim=16, action_dim=8)
agent.build_model()

# Train for 50,000 timesteps
agent.train(env, total_timesteps=50000)

# Save trained model
agent.save("trained_quantum_agent.zip")

    # Parse research papers and generate algorithms
import qmetagpt

# Initialize LLM processor
llm = qmetagpt.LLMProcessor(model_type="openai", model_name="gpt-4")

# Parse arXiv paper
paper_data = qmetagpt.parse_arxiv_paper("2301.12345")

# Extract algorithm from paper
algorithm_pseudocode = llm.extract_pseudocode(paper_data['abstract'])

# Generate quantum implementation
circuit = qmetagpt.synthesize_from_pseudocode(algorithm_pseudocode)

print("📚 Generated algorithm from research paper!")

    Batch Processing

    # Process multiple quantum tasks
tasks = [
    np.random.random(8) for _ in range(10)  # 10 random tasks
]

results = []
for i, task in enumerate(tasks):
    print(f"Processing task {i+1}/10...")
    circuit = agent.generate_circuit(task)
    result = evaluator.evaluate(circuit)
    results.append(result)

# Generate batch report
report_builder.generate_batch_report(results, "batch_analysis.pdf")
