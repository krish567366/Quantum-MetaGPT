    Report Generation

    Introduction

    QuantumMetaGPT's report generation system creates comprehensive research reports, performance analyses, and technical documentation in multiple formats. This guide covers report templates, automated generation, and customization options.

    Report Architecture

    graph TD A[Data Sources] --> B[Report Builder] B --> C[Template Engine] C --> D[Content Generator] D --> E[Format Processors] A1[Evaluation Results] --> A A2[Algorithm Performance] --> A A3[Optimization Data] --> A A4[Research Papers] --> A E --> F[LaTeX/PDF] E --> G[HTML/Web] E --> H[Markdown] E --> I[Jupyter Notebooks] J[Visualization Engine] --> K[Plots & Charts] K --> D L[Statistical Analysis] --> M[Tables & Metrics] M --> D style A fill:#e1f5fe style B fill:#fff3e0 style F fill:#c8e6c9 style G fill:#c8e6c9 style H fill:#c8e6c9 style I fill:#c8e6c9

    Core Components

    The report generation system includes:

    1. Report Builder: Main interface for creating reports
    2. Template Engine: LaTeX and HTML templates
    3. Visualization: Automated plot and chart generation
    4. Statistical Analysis: Performance metrics and comparisons
    5. Multi-Format Export: PDF, HTML, Markdown, Jupyter outputs

    Comprehensive Research Report

    def generate_research_report(experiment_data, output_format="pdf"):
    """Generate comprehensive research report"""

    report = ReportBuilder(
        title=experiment_data['title'],
        author=experiment_data['authors'],
        template="ieee_paper",
        bibliography=True
    )

    # Abstract
    report.add_abstract(experiment_data['abstract'])

    # Introduction
    report.add_section("Introduction", {
        'content': experiment_data['introduction'],
        'subsections': [
            {'title': 'Motivation', 'content': experiment_data['motivation']},
            {'title': 'Contributions', 'content': experiment_data['contributions']}
        ]
    })

    # Methodology
    report.add_section("Methodology", {
        'algorithms': experiment_data['algorithms'],
        'implementation': experiment_data['implementation_details'],
        'experimental_setup': experiment_data['setup']
    })

    # Results
    report.add_results_section(experiment_data['results'])

    # Discussion
    report.add_section("Discussion", {
        'analysis': experiment_data['analysis'],
        'limitations': experiment_data['limitations'],
        'future_work': experiment_data['future_work']
    })

    # References
    if 'references' in experiment_data:
        report.add_bibliography(experiment_data['references'])

    # Generate report
    if output_format == "pdf":
        return report.generate_pdf()
    elif output_format == "html":
        return report.generate_html()
    elif output_format == "notebook":
        return report.generate_jupyter_notebook()
    else:
        return report.generate_markdown()

def analyze_overall_performance(algorithms, test_problems, metrics):
    """Analyze overall performance across all algorithms"""

    results = {}
    for metric in metrics:
        results[metric] = {}
        for algo_name in algorithms.keys():
            # Collect metric values across all test problems
            metric_values = []
            for problem in test_problems:
                value = evaluate_single_metric(algorithms[algo_name], problem, metric)
                metric_values.append(value)

            results[metric][algo_name] = {
                'mean': np.mean(metric_values),
                'std': np.std(metric_values),
                'min': np.min(metric_values),
                'max': np.max(metric_values)
            }

    # Generate summary insights
    insights = {
        'best_overall': find_best_overall_algorithm(results),
        'best_per_metric': {metric: find_best_for_metric(results[metric]) 
                           for metric in metrics},
        'performance_trends': identify_performance_trends(results),
        'significant_differences': find_significant_differences(results)
    }

    return {
        'performance_data': results,
        'insights': insights,
        'methodology': 'Performance evaluated using paired t-tests with Bonferroni correction'
    }

def create_convergence_plot(optimization_history):
    """Create convergence visualization"""

    import matplotlib.pyplot as plt

    fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10, 8))

    # Objective value over iterations
    iterations = optimization_history['iterations']
    objective_values = optimization_history['objective_values']

    ax1.plot(iterations, objective_values, 'b-', linewidth=2)
    ax1.set_xlabel('Iteration')
    ax1.set_ylabel('Objective Value')
    ax1.set_title('Optimization Convergence')
    ax1.grid(True, alpha=0.3)

    # Log-scale improvement
    if len(objective_values) > 1:
        improvements = [abs(objective_values[i] - objective_values[i-1]) 
                       for i in range(1, len(objective_values))]
        ax2.semilogy(iterations[1:], improvements, 'r-', linewidth=2)
        ax2.set_xlabel('Iteration')
        ax2.set_ylabel('Improvement (log scale)')
        ax2.set_title('Convergence Rate')
        ax2.grid(True, alpha=0.3)

    plt.tight_layout()

    # Save plot to temporary file
    import tempfile
    temp_file = tempfile.mktemp(suffix='.png')
    plt.savefig(temp_file, dpi=300, bbox_inches='tight')
    plt.close()

    return temp_file

def analyze_vqe_energies(vqe_results):
    """Analyze VQE energy results"""

    final_energy = vqe_results['final_energy']
    reference_energy = vqe_results.get('reference_energy')
    energy_history = vqe_results['energy_history']

    analysis = {}

    if reference_energy is not None:
        energy_error = abs(final_energy - reference_energy)
        analysis['error_vs_reference'] = energy_error
        analysis['chemical_accuracy_achieved'] = energy_error < 0.0016  # 1 kcal/mol

    # Statistical analysis of energy convergence
    if len(energy_history) > 10:
        final_energies = energy_history[-10:]  # Last 10 values
        analysis['variance'] = np.var(final_energies)
        analysis['confidence_intervals'] = {
            '95%': np.percentile(final_energies, [2.5, 97.5]),
            '99%': np.percentile(final_energies, [0.5, 99.5])
        }

    return analysis

    Creating Custom Templates

    class CustomReportTemplate:
    def __init__(self, template_name):
        self.template_name = template_name
        self.sections = []
        self.style_config = {}
        self.latex_preamble = ""

    def set_style(self, **style_options):
        """Set custom styling options"""
        self.style_config.update(style_options)

    def add_latex_package(self, package_name, options=""):
        """Add LaTeX package to preamble"""
        if options:
            self.latex_preamble += f"\\usepackage[{options}]{{{package_name}}}\n"
        else:
            self.latex_preamble += f"\\usepackage{{{package_name}}}\n"

    def define_section_template(self, section_name, template_string):
        """Define custom section template"""
        self.sections.append({
            'name': section_name,
            'template': template_string
        })

    def save_template(self, filepath):
        """Save template configuration"""
        import json

        template_config = {
            'name': self.template_name,
            'sections': self.sections,
            'style': self.style_config,
            'latex_preamble': self.latex_preamble
        }

        with open(filepath, 'w') as f:
            json.dump(template_config, f, indent=2)

    def use_custom_template(template_path, report_data):
    """Use custom template for report generation"""

    import json

    # Load template configuration
    with open(template_path, 'r') as f:
        template_config = json.load(f)

    # Create report with custom template
    report = ReportBuilder(
        title=report_data['title'],
        template=template_config,
        custom_style=template_config['style']
    )

    # Add sections using custom templates
    for section_config in template_config['sections']:
        section_name = section_config['name']
        if section_name in report_data:
            report.add_custom_section(
                section_config['template'],
                report_data[section_name]
            )

    return report

    def setup_automated_reporting(schedule_config):
    """Set up automated report generation"""

    import schedule
    import time
    from datetime import datetime

    def generate_daily_performance_report():
        """Generate daily performance summary"""

        # Collect performance data
        performance_data = collect_daily_performance_metrics()

        # Generate report
        report = ReportBuilder(
            title=f"Daily Performance Report - {datetime.now().strftime('%Y-%m-%d')}",
            template="daily_summary"
        )

        report.add_section("Performance Summary", performance_data)

        # Save with timestamp
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = f"daily_report_{timestamp}.pdf"

        report.generate_pdf(filename)

        # Send notification (optional)
        send_report_notification(filename, performance_data['summary'])

    def generate_weekly_comparison_report():
        """Generate weekly algorithm comparison"""

        # Collect week's data
        weekly_data = collect_weekly_comparison_data()

        # Generate comprehensive comparison
        comparison_report = generate_comparison_report(
            weekly_data['algorithms'],
            weekly_data['test_problems'],
            weekly_data['metrics']
        )

        # Archive report
        week_number = datetime.now().isocalendar()[1]
        filename = f"weekly_comparison_week_{week_number}.pdf"
        save_archived_report(comparison_report, filename)

    # Schedule reports
    if schedule_config.get('daily_reports', False):
        schedule.every().day.at("08:00").do(generate_daily_performance_report)

    if schedule_config.get('weekly_reports', False):
        schedule.every().monday.at("09:00").do(generate_weekly_comparison_report)

    # Run scheduler
    print("Automated reporting started...")
    while True:
        schedule.run_pending()
        time.sleep(60)  # Check every minute
